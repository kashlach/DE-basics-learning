# mkdir - создаем директорию/каталог/папку

# cd ... - перейдем в созданный каталог

# git init - инициализируем (~создаем) репозиторий в текущей папке

# ls - выводим содержимое текущего каталога (или ls -la для скрытых файлов и папок)
  пока тут будут только всякие системный файлы/папки

# создаем нек-рый файл

# git status - проверяем состояние файла/файлов в рабочем каталоге
  т.к файл только добавлен в проект, его статус - Untracked 

# git add - начать отслеживать файл (добавить под версионный контроль), а также индексация изменений
  git add filename - только указанный файл
  git add * - почти все файлы в текущей папке, без скрытых (.env, .gitignore)
  git add . - все файлы в текущей папке, включая скрытые файлы и папки

# снова выполняем git status и видим, что файл добавлен в индекс
	Changes to be committed:...
		new file: ... 
	
  при изменении существующего файла
	Changes not staged for commit:... - отслеживаемый файл был изменён в рабочем каталоге, но пока не проиндексирован
		modified:  ...

  проиндексировали новый файл и изменения в существующем, выполнили git status и видим
	Changes to be committed:...
		new file:   ...
		modified:   ...
		
  теперь оба файла войдут в следующий коммит

# git commit - фиксируем все ПРОИНДЕКСИРОВАННЫЕ изменения (т.е. файл попадёт в коммит в том состоянии, в котором он находился при последнем выполнении git add)

# git log - смотрим историю коммитов (с ключом --oneline короткая версия)



______Компоненты инфраструктуры______
Рабочая директория - рабочая копия нек-рой версии проекта; файлы, с к-рыми работаем
Индекс (staging area, неофиц. кэш-зона) - файл с фиксацией изменений, подготовленных для добавления в коммит
Репозиторий - папка проекта, отслеживаемого Git, содержащая дерево изменений проекта. Файлы истории хранятся в папке .git/ внутри папки проекта

РД -> git add -> Индекс -> git commit -> Репо



_____Статусы_____
untracked -> git add -> staged -> git commit -> unmodified (commited) -> edit -> modified
                          <------------------------ git add <------------------------
   <------------------- remove <---------------------
  

   
_____Больше про коммит_____
# git show (git show <commit_id>) - показать детальную информацию о последнем (или указанном) коммите

# git add + git commit = git commit -a (a от all) - "ленивый" коммит
  заменяет две команды одной, но в коммит НЕ ПОПАДУТ файлы в статусе UNTRACKED
  Практический пример: 
  Создали новый файл со стилями css, добавили эти стили в отслеживаемые файлы html и сделали ленивый коммит. 
  Результат - нерабочие файлики, т.к. файл .css не будет добавлен в коммит
  Как починить: т.к. каждый коммит должен фиксировать РАБОЧУЮ ВЕРСИЮ (атомарные изменения), нужно внести изменения в предыдущий коммит, а не делать новый, который будет чинить прошлый. git add + git commit --amend
  
# git commit --amend - добавить изменения в индексе в предыдущий коммит

# git reset HEAD~1  - отменить последний коммит и изменения в индексе(?)



_____Сообщение коммита_____
# git commit -m "..." - указать тему сообщения без открытия редактора (m от message)
  git commit -am "..." - коммит всех проиндексированных изменений в отслеживаемых файлах сразу с темой сообщения

Тема сообщения: <= 50 символов, капс, что делает коммит (не зачем/почему/как)
Пустая строка
Тело сообщения: <= 80 символов, что изменено
                   (<= 80 символов - рекомендация по переносу строк, чтобы сообщение нормально читалось в терминале git log. 
				    Само тело может быть хоть на 20 строк, но каждая строка примерно до 72–80 символов)
				   
				   

_____Откат изменений_____
# git reset
  Сценарии использования:
  1. убираем из индекса все, что сейчас staged for commit (changes to be commited) # git reset = git reset --mixed HEAD
  1.2 убираем из индекса конкретный файл # git reset filename = git reset --mixed filename
    --то есть здесь изменения в файлах сохраняются в рабочей дириктории, только исключаем их из индекса
  2. возврат к состоянию последнего коммита # git reset --hard
    --здесь мы откатываем сами изменения в файлах к состоянию предыдущего коммита, безвозвратно(!)
  3. переход к состоянию нек-рого коммита
    # git reset shorthashcode - к состоянию указанного коммита
	--
    # git reset HEAD~1 = git reset HEAD^ - возврат на один коммит назад
	# git reset HEAD~2 = git reset HEAD^^ - на два коммита от последнего
	--перемещаем указатель HEAD на нужную позицию. Журнал перемещений указателя HEAD можно посмотреть по команде git reflog (напрмиер чтобы найти идентификаторы для отката)
	--то есть коммит не удаляется, мы только перемещаем указатель
	--изменения в файлах сохраняются в рабочей директории, меняем состояние репозитория
	--чтобы отменить и изменения в рабочей директории тоже, используем например # git reset HEAD^ --hard
	
Еще раз про ключи:
  soft = затрагивает только репозиторий, изменения из коммита перебросятся в индекс и останутся в рабочей директории
  mixed = сбросим индекс и удалим коммит, но изменения в рабочей папке останутся
    (git reset --mixed HEAD = возврат к коммиту, на к-рый указывает HEAD, с очищением индекса, но сохранением раб. директории)
  hard = удаляет изменения везде (!аккуратно, если изменения не были закоммичены. Если были, то можем восстановить)
  --
  git commit --amend = (git reset --soft HEAD~1 + git commit) = отменяем последний коммит, его изменения помещаем в индекс
