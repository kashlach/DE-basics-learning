# mkdir - создаем директорию/каталог/папку

# cd ... - перейти в созданный каталог

# git init - инициализируем (~создаем) репозиторий в текущей папке

# ls - выводим содержимое текущего каталога (ls -la - + скрытые файлы и папки)
пока тут будут только всякие системный файлы/папки

# создаем нек-рый файл

# git status - проверяем состояние файла/файлов в рабочем каталоге
т.к файл только добавлен в проект, его статус - Untracked 

# git add - начать отслеживать файл (добавить под версионный контроль), а также индексация изменений

# снова выполняем git status и видим, что файл добавлен в индекс
	Changes to be committed:...
		new file: ... 
	
при изменении существующего файла
	Changes not staged for commit:... - отслеживаемый файл был изменён в рабочем каталоге, но пока не проиндексирован
		modified:  ...

проиндексировали новый файл и изменения в существующем, выполнили git status и видим
	Changes to be committed:...
		new file:   ...
		modified:   ...
теперь оба файла войдут в следующий коммит

# git commit - фиксируем все ПРОИНДЕКСИРОВАННЫЕ изменения (т.е. файл попадёт в коммит в том состоянии, в котором он находилсяпри последнем выполнении git add)
